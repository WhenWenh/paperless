package main

import (
        "errors"  // Erstellt einfache Fehlerwerte.
        "net"     // Verbindet Host und Port korrekt.
        "net/url" // Erstellt eine korrekt kodierte Verbindungs-URL.
        "os"      // Liest Umgebungsvariablen.
)

type Config struct {
        RabbitHost     string
        RabbitPort     string
        RabbitUser     string
        RabbitPassword string
        HealthAddress  string
}

func loadConfig() (Config, error) {
        // := legt eine Variable an und leitet ihren Typ automatisch ab.
        config := Config{
                RabbitHost: envOrDefault("RABBITMQ_HOST", "localhost"),

                RabbitPort: envOrDefault("RABBITMQ_PORT", "5672"),

                RabbitUser:     os.Getenv("RABBITMQ_USER"),
                RabbitPassword: os.Getenv("RABBITMQ_PASSWORD"),

                HealthAddress: ":8082",
        }

        if config.RabbitUser == "" || config.RabbitPassword == "" {
                return Config{}, errors.New(
                        "RABBITMQ_USER and RABBITMQ_PASSWORD are required",
                )
        }

        return config, nil
}

func (config Config) rabbitURL() string {
        address := url.URL{
                // Verbindungsprotokoll.
                Scheme: "amqp",

                // Beispielsweise "rabbitmq:5672".
                Host: net.JoinHostPort(
                        config.RabbitHost,
                        config.RabbitPort,
                ),

                // Kodiert auch Sonderzeichen in Zugangsdaten korrekt.
                User: url.UserPassword(
                        config.RabbitUser,
                        config.RabbitPassword,
                ),

                // Standard-Virtual-Host "/".
                Path: "/",
        }

        // URL-Objekt in einen String umwandeln.
        return address.String()
}

// Liest einen Wert oder liefert den angegebenen Standardwert.
func envOrDefault(name, fallback string) string {
        value := os.Getenv(name)

        if value == "" {
                return fallback
        }

        return value
}

package main

import (
        "net"      // TCP-Port öffnen.
        "net/http" // HTTP-Server und Antworten.
        "time"     // Timeout konfigurieren.
)

// ready ist eine übergebene Funktion, die Bereitschaft prüft.
// Rückgabe: Server, Fehlerkanal, möglicher Startfehler.
func startHealthServer(
        address string,
        ready func() bool,
) (*http.Server, <-chan error, error) {
        // Port sofort öffnen, damit Portkonflikte sofort auffallen.
        listener, err := net.Listen("tcp", address)
        if err != nil {
                return nil, nil, err
        }

        // HTTP-Router anlegen.
        mux := http.NewServeMux()

        // Handler für GET /health registrieren.
        mux.HandleFunc("/health", func(
                writer http.ResponseWriter,
                request *http.Request,
        ) {
                // Zustand über die übergebene Funktion prüfen.
                if !ready() {
                        http.Error(
                                writer,
                                "RabbitMQ unavailable",
                                http.StatusServiceUnavailable,
                        )
                        return
                }

                // Antwortformat bekanntgeben.
                writer.Header().Set("Content-Type", "application/json")

                // Ohne vorherigen Fehlerstatus wird HTTP 200 gesendet.
                // _ ignoriert die Anzahl geschriebener Bytes.
                _, _ = writer.Write([]byte(`{"status":"UP"}`))
        })

        // HTTP-Server konfigurieren.
        server := &http.Server{
                Handler: mux,

                // Clients dürfen nicht unbegrenzt lange Header übertragen.
                ReadHeaderTimeout: 5 * time.Second,
        }

        // Gepufferter Kanal für einen Serverfehler.
        serverErrors := make(chan error, 1)

        // Goroutine: Server parallel zum RabbitMQ-Consumer ausführen.
        go func() {
                serverErrors <- server.Serve(listener)
        }()

        return server, serverErrors, nil
}

package main

import (
        "context"   // Gemeinsames Signal zum Beenden.
        "log/slog"  // Strukturiertes Logging.
        "os"        // Standardausgabe und Exit-Code.
        "os/signal" // Betriebssystemsignale empfangen.
        "syscall"   // Enthält SIGTERM für Docker-Stopp.
)

func main() {
        // Logs als JSON auf die Standardausgabe schreiben.
        // Docker sammelt diese Ausgabe für "docker compose logs".
        slog.SetDefault(
                slog.New(slog.NewJSONHandler(os.Stdout, nil)),
        )

        // Context wird beendet, wenn Ctrl+C oder SIGTERM eintrifft.
        ctx, stop := signal.NotifyContext(
                context.Background(),
                os.Interrupt,
                syscall.SIGTERM,
        )

        // Beim regulären Verlassen der Funktion Signalbehandlung aufräumen.
        defer stop()

        // Konfiguration aus config.go laden.
        config, err := loadConfig()

        // Go behandelt erwartbare Fehler über Rückgabewerte.
        if err != nil {
                slog.Error("Invalid configuration", "error", err)

                os.Exit(1)
        }

        // Verbindung, Consumer und Healthcheck starten.
        if err := runWorker(ctx, config); err != nil {
                slog.Error("OCR worker stopped", "error", err)
                os.Exit(1)
        }
}

import (
        "context"       // Ermöglicht Abbruch beim Herunterfahren.
        "encoding/json" // Entspricht hier dem JSON-Konverter.
        "errors"        // Einfache Fehler erzeugen.
        "fmt"           // Fehler mit zusätzlichem Kontext umschließen.
        "log/slog"      // Logging.
        "time"          // Pause vor erneuter Zustellung.

        amqp "github.com/rabbitmq/amqp091-go"
)

func decodeMessage(body []byte) (DocumentUploadedMessage, error) {
        var message DocumentUploadedMessage

        // &message übergibt die Adresse der Struktur.
        // Unmarshal kann dadurch deren Felder befüllen.
        if err := json.Unmarshal(body, &message); err != nil {
                // Keine vollständige Nachricht im Fehlertext ausgeben.
                return message, errors.New("invalid OCR message JSON")
        }

        if message.DocumentID == "" || message.StorageUUID == "" {
                return message, errors.New(
                        "documentId and storageUuid are required",
                )
        }

        if message.ContentType != "application/pdf" {
                return message, errors.New("expected application/pdf")
        }

        return message, nil
}

        )

}

func handleDelivery(
        ctx context.Context,
        delivery amqp.Delivery,
) error {
        message, processingErr := decodeMessage(delivery.Body)

        if processingErr == nil {
                slog.Info(
                        "OCR job received",
                        "documentId", message.DocumentID,
                        "storageUuid", message.StorageUUID,
                        "redelivered", delivery.Redelivered,
                )

                // Ergebnis der Verarbeitung merken.
                processingErr = processDocument(message)
        }

        if processingErr != nil {
                slog.Error(
                        "OCR job failed; returning message to queue",
                        "documentId", message.DocumentID,
                        "storageUuid", message.StorageUUID,
                        "error", processingErr,
                )

                select {
                case <-ctx.Done():
                        return ctx.Err()

                case <-time.After(2 * time.Second):
                        // Nach der Pause weiter zur Rückgabe der Nachricht.
                }

                // false: nur diese Nachricht.
                // true: erneut in die Queue stellen.
                if err := delivery.Nack(false, true); err != nil {
                        return fmt.Errorf("request redelivery: %w", err)
                }

                // Der Verarbeitungsfehler wurde behandelt.
                // Das bedeutet NICHT, dass die OCR erfolgreich war.
                return nil
        }

        // Verarbeitung erfolgreich: einzelne Nachricht bestätigen.
        if err := delivery.Ack(false); err != nil {
                return fmt.Errorf("acknowledge OCR job: %w", err)
        }

        slog.Info(
                "OCR placeholder completed; acknowledgement sent",
                "documentId", message.DocumentID,
                "storageUuid", message.StorageUUID,
        )

        return nil
}

package main

import (
        "context"  // Stoppsignal.
        "errors"   // Einfache Fehlerwerte.
        "fmt"      // Fehler mit Kontext.
        "log/slog" // Logging.

        amqp "github.com/rabbitmq/amqp091-go"
)

func runWorker(ctx context.Context, config Config) error {
        // Verbindung zu RabbitMQ herstellen.
        connection, err := amqp.Dial(config.rabbitURL())
        if err != nil {
                return errors.New(
                        "RabbitMQ connection failed; check broker and configuration",
                )
        }

        // Verbindung beim Verlassen dieser Funktion schließen.
        defer connection.Close()

        // AMQP-Kanal innerhalb der Verbindung öffnen.
        channel, err := connection.Channel()
        if err != nil {
                return fmt.Errorf("open RabbitMQ channel: %w", err)
        }
        defer channel.Close()

        // Entspricht new Queue(QueueNames.DOCUMENT_UPLOAD_QUEUE, true).
        queue, err := channel.QueueDeclare(
                documentUploadQueue, // "document.uploaded"
                true,                // dauerhaft
                false,               // nicht automatisch löschen
                false,               // nicht exklusiv
                false,               // Bestätigung der Deklaration abwarten
                nil,                 // keine zusätzlichen Queue-Argumente
        )
        if err != nil {
                return fmt.Errorf("declare upload queue: %w", err)
        }

        // Maximal eine unbestätigte Nachricht für diesen Consumer.
        if err := channel.Qos(1, 0, false); err != nil {
                return fmt.Errorf("configure prefetch: %w", err)
        }

        // Consumer registrieren.
        deliveries, err := channel.Consume(
                queue.Name, // Queue abonnieren.
                "",         // RabbitMQ erzeugt einen Consumer-Namen.
                false,      // Nicht sofort automatisch bestätigen.
                false,      // Andere Consumer wären grundsätzlich erlaubt.
                false,      // no-local, für RabbitMQ hier false.
                false,      // Registrierung bestätigen lassen.
                nil,        // Keine zusätzlichen Consumer-Argumente.
        )
        if err != nil {
                return fmt.Errorf("register consumer: %w", err)
        }

        // Eigenen Healthcheck starten.
        server, serverErrors, err := startHealthServer(
                config.HealthAddress,

                // Funktion wird bei jedem Healthcheck aufgerufen.
                func() bool {
                        return !connection.IsClosed() && !channel.IsClosed()
                },
        )
        if err != nil {
                return fmt.Errorf("start health server: %w", err)
        }

        // HTTP-Server beim Beenden schließen.
        defer server.Close()

        slog.Info(
                "OCR worker ready",
                "queue", queue.Name,
        )

        // Ohne Bedingung ist for eine Endlosschleife.
        for {
                // Auf eines der folgenden Ereignisse warten.
                select {
                case <-ctx.Done():
                        // Ctrl+C oder Docker-Stopp.
                        slog.Info("OCR worker shutting down")
                        return nil

                case err := <-serverErrors:
                        // Health-Server unerwartet beendet.
                        return fmt.Errorf("health server stopped: %w", err)

                case delivery, open := <-deliveries:
                        // open=false bedeutet: Zustellungsstrom geschlossen.
                        if !open {
                                return errors.New(
                                        "RabbitMQ delivery stream closed",
                                )
                        }

                        // Entspricht dem Aufruf deiner receive(...)-Methode.
                        if err := handleDelivery(ctx, delivery); err != nil {
                                // Angeforderter Stopp ist kein Betriebsfehler.
                                if ctx.Err() != nil {
                                        return nil
                                }

                                // Technischer Fehler, beispielsweise ACK fehlgeschlagen.
                                return err
                        }
                }
        }
}
